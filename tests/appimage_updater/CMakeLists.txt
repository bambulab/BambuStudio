cmake_minimum_required(VERSION 3.16)
project(appimage_updater_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(appimage_updater_tests updater_tests.cpp ../../src/slic3r/Utils/AppImageUpdater.cpp ../../src/slic3r/Utils/ReleaseUpdater.cpp ../../src/slic3r/Utils/DesktopUpdater.cpp)
target_include_directories(appimage_updater_tests PRIVATE ../../src ..)
target_link_libraries(appimage_updater_tests PRIVATE CURL::libcurl OpenSSL::Crypto Threads::Threads)
target_compile_options(appimage_updater_tests PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
add_test(NAME appimage_updater COMMAND appimage_updater_tests)
add_executable(appimage_download_smoke EXCLUDE_FROM_ALL download_smoke.cpp ../../src/slic3r/Utils/AppImageUpdater.cpp ../../src/slic3r/Utils/ReleaseUpdater.cpp ../../src/slic3r/Utils/DesktopUpdater.cpp)
target_include_directories(appimage_download_smoke PRIVATE ../../src)
target_link_libraries(appimage_download_smoke PRIVATE CURL::libcurl OpenSSL::Crypto Threads::Threads)
